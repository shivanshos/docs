## 2. VPS Setup Scripts

### vps/ollama-setup.sh

```bash
#!/bin/bash
# Ollama installation and configuration for ShivanshOS VPS
# Source: https://ollama.com/install.sh

set -e

echo "=== ShivanshOS VPS Ollama Setup ==="

# Update system
apt update && apt upgrade -y

# Install dependencies
apt install -y curl git ufw docker.io docker-compose-plugin nginx certbot

# Create non-root user
if ! id "deploy" &>/dev/null; then
    adduser --disabled-password --gecos "" deploy
    usermod -aG sudo deploy
fi

# Install Ollama
echo "Installing Ollama..."
curl -fsSL https://ollama.com/install.sh | sh

# Configure Ollama to listen on all interfaces
cat > /etc/systemd/system/ollama.service.d/override.conf << 'EOF'
[Service]
Environment="OLLAMA_HOST=0.0.0.0:11434"
Environment="OLLAMA_ORIGINS=*"
EOF

# Pull AI models for agents
echo "Pulling AI models..."
ollama pull llama3.2:3b
ollama pull codestral:7b
ollama pull qwen2.5:3b
ollama pull mistral:7b

# Enable and start Ollama
systemctl daemon-reload
systemctl enable ollama
systemctl start ollama

# Configure firewall
ufw allow 22/tcp
ufw allow 80/tcp
ufw allow 443/tcp
ufw allow 11434/tcp
ufw --force enable

echo "Ollama setup complete!"
echo "API available at: http://$(hostname -I | awk '{print $1}'):11434"
```

## 5. Cloudflare Worker: Agent Dashboard

### cloudflare/workers/agent-dashboard/src/index.ts

```typescript
import { Hono } from 'hono';
import { cors } from 'hono/cors';
import { jwt } from 'hono/jwt';

const app = new Hono();

// CORS for all routes
app.use('*', cors());

// OAuth middleware
app.use('/api/*', async (c, next) => {
  const authHeader = c.req.header('Authorization');
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return c.json({ error: 'Unauthorized' }, 401);
  }
  await next();
});

// Agent configuration endpoint
app.post('/api/agents/configure', async (c) => {
  const config = await c.req.json();
  
  // Store in Cloudflare KV
  await c.env.AGENT_CONFIG.put('infrastructure', JSON.stringify(config));
  
  // Trigger deployment worker
  await c.env.DEPLOYMENT_WORKER.fetch('https://deployment.shivanshsethi.in/trigger', {
    method: 'POST',
    body: JSON.stringify(config)
  });
  
  return c.json({ success: true, message: 'Agent configured' });
});

// OAuth callback
app.get('/oauth/callback', async (c) => {
  const code = c.req.query('code');
  const state = c.req.query('state');
  
  // Exchange code for token
  const tokenResponse = await fetch('https://shivanshos.cloudflareaccess.com/cdn-cgi/access/callback', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      code,
      client_id: c.env.OAUTH_CLIENT_ID,
      client_secret: c.env.OAUTH_CLIENT_SECRET,
      redirect_uri: 'https://dash.shivanshsethi.in/oauth/callback'
    })
  });
  
  const tokens = await tokenResponse.json();
  
  // Store session
  c.set('jwt', tokens.access_token);
  
  return c.redirect('/dashboard');
});

// Dashboard endpoint
app.get('/dashboard', async (c) => {
  const config = await c.env.AGENT_CONFIG.get('infrastructure', { type: 'json' });
  
  return c.json({
    infrastructure: config,
    ollama_status: 'active',
    cloudflare_status: 'connected'
  });
});

export default app;
```

````text
## 7. Setup Scripts

### scripts/setup-ollama.sh

```bash
#!/bin/bash
# Complete Ollama + Open WebUI setup

VPS_IP="<your-vps-ip>"
DOMAIN="vps.shivanshsethi.in"

# SSH into VPS and execute setup
ssh root@$VPS_IP << 'EOF'
# Install Ollama
curl -fsSL https://ollama.com/install.sh | sh

# Configure systemd override
mkdir -p /etc/systemd/system/ollama.service.d
cat > /etc/systemd/system/ollama.service.d/override.conf << 'OVERRIDE'
[Service]
Environment="OLLAMA_HOST=0.0.0.0:11434"
Environment="OLLAMA_ORIGINS=*"
OVERRIDE

# Pull models
ollama pull llama3.2:3b
ollama pull codestral:7b
ollama pull qwen2.5:3b

# Restart Ollama
systemctl daemon-reload
systemctl restart ollama

# Install Docker
apt install -y docker.io docker-compose-plugin

# Deploy Open WebUI
docker run -d \
  --name open-webui \
  -p 3000:8080 \
  -v open-webui:/app/backend/data \
  -e OLLAMA_BASE_URL=http://localhost:11434 \
  --restart always \
  ghcr.io/open-webui/open-webui:main

# Configure Nginx
apt install -y nginx certbot
cat > /etc/nginx/sites-available/$DOMAIN << 'NGINX'
server {
    listen 80;
    server_name vps.shivanshsethi.in;
    
    location / {
        proxy_pass http://localhost:3000;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
    }
    
    location /ollama/ {
        proxy_pass http://localhost:11434/;
        proxy_set_header Host \$host;
    }
}
NGINX

ln -s /etc/nginx/sites-available/$DOMAIN /etc/nginx/sites-enabled/
nginx -t
systemctl reload nginx

# SSL certificate
certbot --nginx -d $DOMAIN --non-interactive --agree-tos --email admin@shivanshsethi.in

echo "Setup complete!"
EOF
````

### scripts/configure-cloudflare-access.sh

```bash
#!/bin/bash
# Cloudflare Access & OAuth configuration

CLOUDFLARE_API_TOKEN="<your-api-token>"
ZONE_ID="<your-zone-id>"
ACCOUNT_ID="<your-account-id>"

# Create Access Application
curl -X POST "https://api.cloudflare.com/client/v4/accounts/$ACCOUNT_ID/access/apps" \
  -H "Authorization: Bearer $CLOUDFLARE_API_TOKEN" \
  -H "Content-Type: application/json" \
  --data '{
    "name": "ShivanshOS Agent Dashboard",
    "domain": "dash.shivanshsethi.in",
    "session_duration": "24h",
    "oauth_configuration": {
      "enabled": true,
      "dynamic_client_registration": {
        "allowed_uris": ["https://dash.shivanshsethi.in/*"]
      },
      "grant": {
        "session_duration": "24h",
        "access_token_lifetime": "15m"
      }
    }
  }'

# Create Access Policy
curl -X POST "https://api.cloudflare.com/client/v4/accounts/$ACCOUNT_ID/access/policies" \
  -H "Authorization: Bearer $CLOUDFLARE_API_TOKEN" \
  -H "Content-Type: application/json" \
  --data '{
    "name": "ShivanshOS Team SSO",
    "application_id": "<app-id-from-previous-step>",
    "decision": "allow",
    "include": [
      {
        "email": {
          "operator": "ends_with",
          "value": "@shivanshsethi.in"
        }
      }
    ]
  }'

echo "Cloudflare Access configured!"
```
